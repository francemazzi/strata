cmake_minimum_required(VERSION 3.22)
project(StrataDiscoveryQt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
enable_testing()
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network Widgets Test Concurrent)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBZIP REQUIRED IMPORTED_TARGET libzip)
set(SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../src/app/ai/discovery")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/qgis_app.h" "#pragma once\n#define APP_EXPORT\n")
add_library(discovery_qt STATIC
  ${SOURCE_ROOT}/qgsaidiscoveryclient.cpp ${SOURCE_ROOT}/qgsaidiscoveryfiles.cpp
  ${SOURCE_ROOT}/qgsaidiscoverydownload.cpp ${SOURCE_ROOT}/qgsaidiscoverypreview.cpp)
target_include_directories(discovery_qt PUBLIC ${SOURCE_ROOT} ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(discovery_qt PUBLIC Qt6::Network Qt6::Widgets PkgConfig::LIBZIP)
add_executable(test_discovery test_discovery.cpp)
target_link_libraries(test_discovery PRIVATE discovery_qt Qt6::Test)
add_test(NAME discovery_qt COMMAND test_discovery)
set_tests_properties(discovery_qt PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
